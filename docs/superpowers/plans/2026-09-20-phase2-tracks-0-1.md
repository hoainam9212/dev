# Giai đoạn 2 – Lộ trình 0 (Làm quen máy tính) + Lộ trình 1 (Gõ phím) Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Thêm 3 trình mô phỏng thật (phòng tập chuột, Desktop + Explorer/Finder cho Windows và Mac, gõ phím có Telex/VNI), bài học cho lộ trình 0 và 1, khu "Luyện gõ" (kiểm tra tốc độ, trò chơi Mèo bắt cá, thống kê), và xử lý các việc còn tồn của giai đoạn 1.

**Architecture:** Mỗi trình mô phỏng gồm hai phần. Phần logic nằm trong `shared/src/sims/*.ts`: reducer thuần, grader, hàm tạo trạng thái ban đầu, có test. Phần giao diện nằm trong `client/src/sims/<sim>/`: React component điều khiển trạng thái **chỉ** qua `useReducer(def.reduce, initial)` và báo ra ngoài bằng `onState`. Nhờ vậy test nội dung, vốn phát lại `solution` qua reducer, chứng minh được bài học qua được trên đúng logic mà giao diện đang dùng. Grader đăng ký vào `GRADERS` (shared), component đăng ký vào `SIMS` (client). Khu "Luyện gõ" dùng lại component gõ phím ở chế độ độc lập và lưu kết quả vào Dexie (`typingRuns`, `keyStats`).

**Tech Stack:** giữ nguyên giai đoạn 1: TypeScript 5.6, React 18.3, react-router-dom 6, zustand, Dexie 4, Tailwind 4, Vitest 2, Testing Library. Không thêm thư viện mới.

```ts
export function useCompleted(): { completed: Set<string>; stars: Record<string, number>; loaded: boolean } {
  const raw = useLiveQuery(() => db.progress.toArray(), []);
  const rows = raw ?? NO_PROGRESS;
  const completed = new Set(rows.filter((r) => r.stars > 0).map((r) => r.lessonId));

  return { completed, stars, loaded: raw !== undefined };
}
```

```ts
export type Finger = 'lp' | 'lr' | 'lm' | 'li' | 'ri' | 'rm' | 'rr' | 'rp' | 'th';

export interface KeyDef {
  id: string;
  label: string;
  shiftLabel?: string;
  finger: Finger;
  home?: boolean;
}

export const FINGER_VI: Record<Finger, string> = {
  lp: 'Ngón út trái',
  lr: 'Ngón áp út trái',
  lm: 'Ngón giữa trái',
  li: 'Ngón trỏ trái',
  ri: 'Ngón trỏ phải',
  rm: 'Ngón giữa phải',
  rr: 'Ngón áp út phải',
  rp: 'Ngón út phải',
  th: 'Ngón cái',
};

/** Màu pastel cho từng ngón (tay trái và phải đối xứng). */
export const FINGER_COLOR: Record<Finger, string> = {
  lp: '#FFC9DE', lr: '#FFE0B3', lm: '#FFF3B0', li: '#C8F2DC',
  ri: '#C6E7FF', rm: '#FFF3B0', rr: '#FFE0B3', rp: '#FFC9DE',
  th: '#E4D9FF',
};

const FINGER_OF: Record<string, Finger> = {};
const assign = (keys: string, f: Finger) => [...keys].forEach((k) => (FINGER_OF[k] = f));
assign('`1qaz', 'lp');
assign('2wsx', 'lr');
assign('3edc', 'lm');
assign('45rtfgvb', 'li');
assign('67yuhjnm', 'ri');
assign('8ik,', 'rm');
assign('9ol.', 'rr');
assign("0-=p[]\\;'/", 'rp');

const HOME = new Set([...'asdfjkl;']);

function row(ids: string, shifted: string): KeyDef[] {
  const a = [...ids];
  const b = [...shifted];
  return a.map((id, i) => ({
    id,
    label: /[a-z]/.test(id) ? id.toUpperCase() : id,
    shiftLabel: /[a-z]/.test(id) ? undefined : b[i],
    finger: FINGER_OF[id] ?? 'rp',
    home: HOME.has(id) || undefined,
  }));
}

export const KEY_ROWS: KeyDef[][] = [
  row('`1234567890-=', '~!@#$%^&*()_+'),
  row('qwertyuiop[]\\', 'QWERTYUIOP{}|'),
  row("asdfghjkl;'", 'ASDFGHJKL:"'),
  row('zxcvbnm,./', 'ZXCVBNM<>?'),
];

export const SPACE_KEY: KeyDef = { id: 'space', label: 'Space', finger: 'th' };

const BY_ID = new Map<string, KeyDef>();
const BY_CHAR = new Map<string, { id: string; shift: boolean }>();
for (const r of KEY_ROWS) {
  for (const k of r) {
    BY_ID.set(k.id, k);
    BY_CHAR.set(k.id, { id: k.id, shift: false });
    if (k.shiftLabel) BY_CHAR.set(k.shiftLabel, { id: k.id, shift: true });
    else BY_CHAR.set(k.id.toUpperCase(), { id: k.id, shift: true });
  }
}
BY_ID.set('space', SPACE_KEY);
```

```ts
import type { CheckResult } from '../grading/core';

export type MouseGame = 'click' | 'double' | 'right' | 'drag' | 'scroll';
export interface MouseState {
  game: MouseGame;
  total: number;
  hitIds: string[];
  misses: number;
}
export type MouseAction = { type: 'hit'; id: string } | { type: 'miss' };
export interface MouseCheck {
  kind: 'mouseGame';
  minHits: number;
  maxMisses?: number;
}

export function mouseInitial(game: MouseGame, total: number): MouseState {
  return { game, total, hitIds: [], misses: 0 };
}

export function mouseTargets(s: MouseState): string[] {
  return Array.from({ length: s.total }, (_, i) => `t${i + 1}`);
}

export function reduceMouse(s: MouseState, a: MouseAction): MouseState {
  if (a.type === 'miss') return { ...s, misses: s.misses + 1 };
  if (s.hitIds.includes(a.id) || !mouseTargets(s).includes(a.id)) return s;
  return { ...s, hitIds: [...s.hitIds, a.id] };
}

export function gradeMouse(s: MouseState, c: MouseCheck): CheckResult {
  const hits = s.hitIds.length;
  if (hits < c.minHits) return { ok: false, messageVi: `Cần trúng ít nhất ${c.minHits} mục tiêu (bạn mới trúng ${hits}).` };
  if (c.maxMisses !== undefined && s.misses > c.maxMisses) {
    return { ok: false, messageVi: `Trượt hơi nhiều (${s.misses} lần) — thử chậm lại một chút nhé.` };
  }
  return { ok: true, messageVi: `Bạn đã trúng ${hits}/${c.minHits} mục tiêu.` };
}
```

```ts
import type { CheckResult } from '../grading/core';
import type { OS } from '../types';
import * as vfs from '../vfs';

type Vfs = vfs.Vfs;

export const UNDO_LIMIT = 30;

export interface DesktopState {
  os: OS;
  fs: Vfs;
  cwd: string;
  selected: string[];
  clipboard: { ids: string[]; mode: 'copy' | 'cut' } | null;
  past: Vfs[];
  error: vfs.VfsError | null;
}

export type DesktopAction =
  | { type: 'open'; id: string }
  | { type: 'up' }
  | { type: 'select'; ids: string[] }
  | { type: 'mkdir'; name: string }
  | { type: 'rename'; id: string; name: string }
  | { type: 'remove'; ids: string[] }
  | { type: 'restore'; id: string }
  | { type: 'emptyTrash' }
  | { type: 'copy'; ids: string[] }
  | { type: 'cut'; ids: string[] }
  | { type: 'paste' }
  | { type: 'move'; ids: string[]; destId: string }
  | { type: 'undo' }
  | { type: 'clearError' };

export type DesktopCheck =
  | { kind: 'pathExists'; path: string; type?: vfs.NodeType }
  | { kind: 'notExists'; path: string }
  | { kind: 'inTrash'; name: string }
  | { kind: 'cwdIs'; path: string };

export function idAt(fs: Vfs, path: string): string {
  const n = vfs.findByPath(fs, path);
  if (!n) throw new Error(`idAt: no node at ${path}`);
  return n.id;
}

export function desktopInitial(tree: vfs.TreeSpec[], os: OS, cwdPath = '/'): DesktopState {
  const fs = vfs.fromTree(tree);
  return { os, fs, cwd: idAt(fs, cwdPath), selected: [], clipboard: null, past: [], error: null };
}

export function uniqueName(fs: Vfs, parentId: string, base: string): string {
  const taken = new Set(vfs.childrenOf(fs, parentId).map((n) => n.name.toLowerCase()));
  if (!taken.has(base.toLowerCase())) return base;
  for (let i = 2; ; i++) {
    const name = `${base} (${i})`;
    if (!taken.has(name.toLowerCase())) return name;
  }
}

/** Áp một chuỗi thao tác vfs; dừng ở lỗi đầu tiên. */
function applyAll(s: DesktopState, ops: ((fs: Vfs) => vfs.VfsResult)[], after?: (fs: Vfs, ids: string[]) => Partial<DesktopState>): DesktopState {
  let fs = s.fs;
  const ids: string[] = [];
  for (const op of ops) {
    const r = op(fs);
    if (!r.ok) return { ...s, error: r.error };
    fs = r.fs;
    ids.push(r.id);
  }
  const cwd = fs.nodes[s.cwd] && !vfs.isInside(fs, s.cwd, fs.trashId) ? s.cwd : s.cwd === fs.trashId ? s.cwd : fs.rootId;
  return {
    ...s,
    fs,
    cwd,
    error: null,
    past: [...s.past, s.fs].slice(-UNDO_LIMIT),
    selected: s.selected.filter((id) => fs.nodes[id]),
    ...(after ? after(fs, ids) : {}),
  };
}

export function reduceDesktop(s: DesktopState, a: DesktopAction): DesktopState {
  switch (a.type) {
    case 'open': {
      const n = s.fs.nodes[a.id];
      if (!n || n.type !== 'folder') return s;
      return { ...s, cwd: a.id, selected: [] };
    }
    case 'up': {
      const n = s.fs.nodes[s.cwd];
      if (!n || n.parentId === null) return s.cwd === s.fs.trashId ? { ...s, cwd: s.fs.rootId, selected: [] } : s;
      return { ...s, cwd: n.parentId, selected: [] };
    }
    case 'select':
      return { ...s, selected: a.ids.filter((id) => s.fs.nodes[id]) };
    case 'mkdir':
      return applyAll(s, [(fs) => vfs.mkdir(fs, s.cwd, a.name)], (_fs, ids) => ({ selected: ids }));
    case 'rename':
      return applyAll(s, [(fs) => vfs.rename(fs, a.id, a.name)]);
    case 'remove':
      return applyAll(s, a.ids.map((id) => (fs: Vfs) => vfs.remove(fs, id)), () => ({ selected: [] }));
    case 'restore':
      return applyAll(s, [(fs) => vfs.restore(fs, a.id)], () => ({ selected: [] }));
    case 'emptyTrash': {
      const fs = vfs.emptyTrash(s.fs);
      return { ...s, fs, past: [...s.past, s.fs].slice(-UNDO_LIMIT), selected: s.selected.filter((id) => fs.nodes[id]), error: null };
    }
    case 'copy':
    case 'cut':
      return { ...s, clipboard: { ids: [...a.ids], mode: a.type } };
    case 'paste': {
      const clip = s.clipboard;
      if (!clip) return s;
      const ops = clip.ids.map((id) =>
        clip.mode === 'cut' ? (fs: Vfs) => vfs.move(fs, id, s.cwd) : (fs: Vfs) => vfs.copy(fs, id, s.cwd, s.os),
      );
      return applyAll(s, ops, (_fs, ids) => ({ selected: ids, clipboard: clip.mode === 'cut' ? null : clip }));
    }
    case 'move':
      return applyAll(s, a.ids.map((id) => (fs: Vfs) => vfs.move(fs, id, a.destId)), () => ({ selected: [] }));
    case 'undo': {
      const prev = s.past[s.past.length - 1];
      if (!prev) return s;
      const cwd = prev.nodes[s.cwd] ? s.cwd : prev.rootId;
      return { ...s, fs: prev, past: s.past.slice(0, -1), cwd, selected: [], error: null };
    }
    case 'clearError':
      return { ...s, error: null };
  }
}

export function gradeDesktop(s: DesktopState, c: DesktopCheck): CheckResult {
  switch (c.kind) {
    case 'pathExists': {
      const n = vfs.findByPath(s.fs, c.path);
      const ok = !!n && (!c.type || n.type === c.type);
      const what = c.type === 'folder' ? 'thư mục' : c.type === 'file' ? 'file' : 'mục';
      return { ok, messageVi: ok ? `Đã có ${what} ${c.path}.` : `Chưa thấy ${what} ${c.path}.` };
    }
    case 'notExists': {
      const ok = !vfs.findByPath(s.fs, c.path);
      return { ok, messageVi: ok ? `${c.path} đã được dọn đi.` : `${c.path} vẫn còn ở đó.` };
    }
    case 'inTrash': {
      const key = c.name.toLowerCase();
      const ok = vfs.childrenOf(s.fs, s.fs.trashId).some((n) => n.name.toLowerCase() === key);
      return { ok, messageVi: ok ? `“${c.name}” đang nằm trong thùng rác.` : `Chưa thấy “${c.name}” trong thùng rác.` };
    }
    case 'cwdIs': {
      const here = vfs.pathOf(s.fs, s.cwd);
      const ok = here.toLowerCase() === c.path.toLowerCase();
      return { ok, messageVi: ok ? `Bạn đang mở ${c.path}.` : `Hãy mở ${c.path} (bạn đang ở ${here}).` };
    }
  }
}
```

```ts
import type { CheckResult } from '../grading/core';
import { emptyIme, imeBackspace, imeInput, imeText, type ImeMethod, type ImeState } from '../ime';
import { accuracy, compareText, countCorrect, recordKey, wpm, type KeyStats } from '../typing';

export type TypingMode = 'drill' | 'words' | 'sentences' | 'passage' | 'test';

export interface TypingState {
  target: string;
  mode: TypingMode;
  ime: ImeMethod | null;
  allowBackspace: boolean;
  durationSec: number | null;
  typed: string;
  imeState: ImeState;
  keystrokes: number;
  correctKeys: number;
  startedAt: number | null;
  lastAt: number | null;
  finished: boolean;
  keyStats: KeyStats;
}

export type TypingAction = { type: 'key'; key: string; at: number } | { type: 'tick'; at: number };

export interface TypingCheck {
  kind: 'typing';
  minWpm?: number;
  minAccuracy?: number;
  completeText?: boolean;
}

export function typingInitial(o: {
  target: string;
  mode?: TypingMode;
  ime?: ImeMethod | null;
  allowBackspace?: boolean;
  durationSec?: number | null;
}): TypingState {
  return {
    target: o.target,
    mode: o.mode ?? 'passage',
    ime: o.ime ?? null,
    allowBackspace: o.allowBackspace ?? true,
    durationSec: o.durationSec ?? null,
    typed: '',
    imeState: emptyIme(),
    keystrokes: 0,
    correctKeys: 0,
    startedAt: null,
    lastAt: null,
    finished: false,
    keyStats: {},
  };
}

export function typedText(s: TypingState): string {
  return s.ime ? imeText(s.imeState) : s.typed;
}

function wrongCommitted(target: string, ime: ImeState): number {
  const t = [...target];
  return [...ime.done].filter((ch, i) => t[i] !== ch).length;
}

function timeUp(s: TypingState, at: number): boolean {
  return s.durationSec !== null && s.startedAt !== null && at - s.startedAt >= s.durationSec * 1000;
}

export function reduceTyping(s: TypingState, a: TypingAction): TypingState {
  if (s.finished) return s;
  if (a.type === 'tick') return timeUp(s, a.at) ? { ...s, finished: true, lastAt: s.startedAt! + s.durationSec! * 1000 } : s;
  if (timeUp(s, a.at)) return { ...s, finished: true, lastAt: s.startedAt! + s.durationSec! * 1000 };

  if (a.key === 'Backspace') {
    if (!s.allowBackspace) return s;
    return s.ime ? { ...s, imeState: imeBackspace(s.imeState) } : { ...s, typed: [...s.typed].slice(0, -1).join('') };
  }
  if ([...a.key].length !== 1) return s;

  const targetChars = [...s.target];
  let next: TypingState;
  let ok: boolean;
  if (s.ime) {
    const imeState = imeInput(s.imeState, a.key, s.ime);
    if ([...imeText(imeState)].length > targetChars.length) return s;
    ok = wrongCommitted(s.target, imeState) <= wrongCommitted(s.target, s.imeState);
    next = { ...s, imeState };
  } else {
    const typedChars = [...s.typed];
    if (typedChars.length >= targetChars.length) return s;
    ok = a.key === targetChars[typedChars.length];
    next = { ...s, typed: s.typed + a.key };
  }
  const startedAt = s.startedAt ?? a.at;
  const done = typedText(next) === s.target;
  return {
    ...next,
    startedAt,
    lastAt: a.at,
    keystrokes: s.keystrokes + 1,
    correctKeys: s.correctKeys + (ok ? 1 : 0),
    keyStats: recordKey(s.keyStats, a.key, ok),
    finished: done,
  };
}

export function typingMetrics(s: TypingState): { wpm: number; accuracy: number; correctChars: number; elapsedMs: number; complete: boolean } {
  const text = typedText(s);
  const correctChars = countCorrect(s.target, text);
  let elapsedMs = s.startedAt !== null && s.lastAt !== null ? s.lastAt - s.startedAt : 0;
  if (s.durationSec !== null) elapsedMs = Math.min(elapsedMs, s.durationSec * 1000);
  return {
    wpm: wpm(correctChars, elapsedMs),
    accuracy: accuracy(s.correctKeys, s.keystrokes),
    correctChars,
    elapsedMs,
    complete: text === s.target,
  };
}

export function gradeTyping(s: TypingState, c: TypingCheck): CheckResult {
  const m = typingMetrics(s);
  const parts: string[] = [];
  let ok = true;
  if ((c.minWpm !== undefined || c.minAccuracy !== undefined) && !s.finished) {
    return { ok: false, messageVi: 'Hãy gõ hết bài để Mochi tính tốc độ nhé.' };
  }
  if (c.minWpm !== undefined) {
    ok = ok && m.wpm >= c.minWpm;
    parts.push(`tốc độ ${m.wpm} WPM (cần ≥ ${c.minWpm})`);
  }
  if (c.minAccuracy !== undefined) {
    ok = ok && m.accuracy >= c.minAccuracy;
    parts.push(`độ chính xác ${m.accuracy}% (cần ≥ ${c.minAccuracy}%)`);
  }
  if (c.completeText) {
    ok = ok && m.complete;
    parts.push(m.complete ? 'đã gõ đúng hết' : `còn ${compareText(s.target, typedText(s)).filter((x) => x !== 'correct').length} ký tự chưa đúng`);
  }
  const text = parts.join(', ');
  return { ok, messageVi: text.charAt(0).toUpperCase() + text.slice(1) + '.' };
}
```

```ts
export interface Fish {
  id: number;
  word: string;
  x: number;
  y: number;
}

export interface FishState {
  words: string[];
  level: number;
  lives: number;
  score: number;
  caught: number;
  fish: Fish[];
  input: string;
  nextId: number;
  spawnIn: number;
  seed: number;
  over: boolean;
}

export type FishAction = { type: 'tick'; dt: number } | { type: 'key'; key: string };

export const fallSpeed = (level: number) => 0.00006 + 0.00002 * (level - 1);
export const spawnInterval = (level: number) => Math.max(900, 2600 - 250 * (level - 1));

function rand(seed: number): [number, number] {
  const next = (Math.imul(seed, 1664525) + 1013904223) >>> 0;
  return [next / 2 ** 32, next];
}

export function fishInitial(words: string[], seed = 1): FishState {
  return { words, level: 1, lives: 3, score: 0, caught: 0, fish: [], input: '', nextId: 1, spawnIn: 0, seed, over: false };
}

export function targetFish(s: FishState): Fish | undefined {
  if (!s.input) return undefined;
  return s.fish.filter((f) => f.word.startsWith(s.input)).sort((a, b) => b.y - a.y)[0];
}

function spawn(s: FishState): FishState {
  const [r1, s1] = rand(s.seed);
  const [r2, s2] = rand(s1);
  const word = s.words[Math.floor(r1 * s.words.length)] ?? 'meo';
  const fish: Fish = { id: s.nextId, word, x: 0.08 + r2 * 0.84, y: 0 };
  return { ...s, fish: [...s.fish, fish], nextId: s.nextId + 1, seed: s2, spawnIn: spawnInterval(s.level) };
}

export function reduceFish(s: FishState, a: FishAction): FishState {
  if (s.over) return s;
  if (a.type === 'tick') {
    const target = targetFish(s);
    const moved = s.fish.map((f) => ({ ...f, y: f.y + fallSpeed(s.level) * a.dt }));
    const landed = moved.filter((f) => f.y >= 1);
    const lives = s.lives - landed.length;
    let next: FishState = {
      ...s,
      fish: moved.filter((f) => f.y < 1),
      lives,
      input: target && landed.some((f) => f.id === target.id) ? '' : s.input,
      spawnIn: s.spawnIn - a.dt,
      over: lives <= 0,
    };
    if (!next.over && next.spawnIn <= 0) next = spawn(next);
    return next;
  }
  if (a.key === 'Backspace') return { ...s, input: s.input.slice(0, -1) };
  if (a.key === 'Enter' || a.key === 'Escape') return { ...s, input: '' };
  if (a.key.length !== 1 || !/[a-z0-9]/i.test(a.key)) return s;
  const input = s.input + a.key.toLowerCase();
  const hit = s.fish.find((f) => f.word === input);
  if (!hit) return { ...s, input };
  const caught = s.caught + 1;
  return {
    ...s,
    input: '',
    fish: s.fish.filter((f) => f.id !== hit.id),
    caught,
    score: s.score + hit.word.length * 10 * s.level,
    level: 1 + Math.floor(caught / 8),
  };
}
```

  - `double`: hộp quà 🎁; `onDoubleClick` → `hit`. Bấm 1 lần thì hiện nhắc "Nhấp nhanh 2 lần liên tiếp nhé!". Bấm nền → `miss`.
  - `right`: thùng 📦; `onContextMenu` (chặn menu trình duyệt) mở menu nhỏ gồm "Mở", "🎁 Mở quà", "Đổi tên". Chọn "🎁 Mở quà" → `hit`, chọn mục khác → `miss`. Bấm trái lên thùng thì hiện nhắc; lời nhắc khác nhau theo `os` (Mac: "giữ phím control rồi bấm, hoặc chạm hai ngón").
  - `drag`: cá 🐟 `draggable`; thả vào giỏ 🧺 (`data-help-id="mouse.basket"`) → `hit`, thả ra ngoài giỏ → `miss`. Id của cá đang kéo lưu trong `useRef` (jsdom không có DataTransfer). Nếu có `e.dataTransfer` thì vẫn `setData` để Firefox chịu kéo.
  - `scroll`: khung cuộn cao 18rem, nội dung cao 1600px; mèo 🐱 thứ `i` nằm ở `top: 360 + i*380 px`. Bấm mèo → `hit`. Ở trò này bấm nền không tính trượt.
- Vị trí mục tiêu cố định theo chỉ số: `left = 8 + (i*37) % 76 %`, `top = 10 + (i*53) % 66 %`.
- Dòng tiến độ `"Đã trúng X/Y"` có `aria-live="polite"`. Khi trúng đủ: Mochi vui + "Tuyệt vời! 🎉".

```tsx
import { useEffect, useReducer, useRef, useState, type MouseEvent as ReactMouseEvent } from 'react';
import { mouseSim } from '@office-neko/shared';
import type { SimProps } from '../types';
import { Mochi } from '../../components/Mochi';

type S = mouseSim.MouseState;

const pos = (i: number) => ({ left: `${8 + ((i * 37) % 76)}%`, top: `${10 + ((i * 53) % 66)}%` });

const LABEL: Record<mouseSim.MouseGame, string> = {
  click: 'Bong bóng',
  double: 'Hộp quà',
  right: 'Thùng',
  drag: 'Cá',
  scroll: 'Mèo',
};
const ICON: Record<mouseSim.MouseGame, string> = { click: '🫧', double: '🎁', right: '📦', drag: '🐟', scroll: '🐱' };

export function MouseSim({ initial, onState, os }: SimProps<S>) {
  const [state, dispatch] = useReducer(mouseSim.reduceMouse, initial);
  const [tip, setTip] = useState<string | null>(null);
  const [menuFor, setMenuFor] = useState<string | null>(null);
  const dragging = useRef<string | null>(null);

  useEffect(() => {
    onState(state);
  }, [state, onState]);

  const targets = mouseSim.mouseTargets(state);
  const remaining = targets.filter((id) => !state.hitIds.includes(id));
  const firstId = remaining[0];
  const done = remaining.length === 0;
  const game = state.game;

  const hit = (id: string) => {
    setTip(null);
    dispatch({ type: 'hit', id });
  };
  const miss = () => dispatch({ type: 'miss' });
  const onBackground = (e: ReactMouseEvent) => {
    if (e.target !== e.currentTarget) return;
    setMenuFor(null);
    if (game === 'click' || game === 'double') miss();
  };

  const itemClass = 'absolute grid h-14 w-14 place-items-center rounded-full text-4xl transition hover:scale-110';

  return (
    <div className="card space-y-3 p-4">
      <p aria-live="polite" className="font-bold">
        Đã trúng {state.hitIds.length}/{state.total}
      </p>
      {game === 'scroll' ? (
        <div data-testid="mouse-area" className="relative h-72 overflow-y-auto rounded-3xl bg-sky/20" onClick={onBackground}>
          <div className="relative" style={{ height: 1600 }}>
            <p className="p-4 text-muted">Lăn bánh xe chuột (hoặc vuốt hai ngón) để tìm các chú mèo đang trốn 👇</p>
            {remaining.map((id) => {
              const i = targets.indexOf(id);
              return (
                <button
                  key={id}
                  type="button"
                  aria-label={`Mèo số ${i + 1}`}
                  data-help-id={id === firstId ? 'mouse.target' : undefined}
                  className={itemClass}
                  style={{ left: `${15 + ((i * 41) % 60)}%`, top: 360 + i * 380 }}
                  onClick={() => hit(id)}
                >
                  🐱
                </button>
              );
            })}
          </div>
        </div>
      ) : (
        <div
          data-testid="mouse-area"
          className="relative h-72 overflow-hidden rounded-3xl bg-sky/20"
          onClick={onBackground}
          onDragOver={(e) => e.preventDefault()}
          onDrop={(e) => {
            if (e.target !== e.currentTarget) return;
            if (dragging.current) miss();
            dragging.current = null;
          }}
        >
          {remaining.map((id) => {
            const i = targets.indexOf(id);
            const common = {
              'data-help-id': id === firstId ? 'mouse.target' : undefined,
              style: pos(i),
            };
            if (game === 'drag') {
              return (
                <span
                  key={id}
                  role="img"
                  aria-label={`Cá số ${i + 1}`}
                  draggable
                  {...common}
                  className={`${itemClass} cursor-grab`}
                  onDragStart={(e) => {
                    dragging.current = id;
                    e.dataTransfer?.setData('text/plain', id);
                  }}
                >
                  🐟
                </span>
              );
            }
            return (
              <button
                key={id}
                type="button"
                aria-label={`${LABEL[game]} số ${i + 1}`}
                {...common}
                className={itemClass}
                onClick={() => {
                  if (game === 'click') hit(id);
                  else if (game === 'double') setTip('Nhấp nhanh 2 lần liên tiếp nhé!');
                  else if (game === 'right')
                    setTip(os === 'mac' ? 'Nhấp chuột phải nhé: giữ phím control rồi bấm, hoặc chạm hai ngón lên trackpad.' : 'Hãy bấm nút bên PHẢI của chuột nhé.');
                }}
                onDoubleClick={() => game === 'double' && hit(id)}
                onContextMenu={(e) => {
                  if (game !== 'right') return;
                  e.preventDefault();
                  setMenuFor(id);
                }}
              >
                {ICON[game]}
              </button>
            );
          })}
          {game === 'drag' && (
            <div
              aria-label="Giỏ cá"
              data-help-id="mouse.basket"
              className="absolute bottom-3 right-3 grid h-24 w-28 place-items-center rounded-3xl bg-butter/70 text-5xl"
              onDragOver={(e) => e.preventDefault()}
              onDrop={(e) => {
                e.stopPropagation();
                if (dragging.current) hit(dragging.current);
                dragging.current = null;
              }}
            >
              🧺
            </div>
          )}
          {menuFor && (
            <ul role="menu" className="card absolute left-1/2 top-1/2 z-10 w-44 -translate-x-1/2 -translate-y-1/2 p-2">
              {['Mở', '🎁 Mở quà', 'Đổi tên'].map((label) => (
                <li key={label}>
                  <button
                    type="button"
                    role="menuitem"
                    className="w-full rounded-xl px-3 py-2 text-left hover:bg-peach/30"
                    onClick={(e) => {
                      e.stopPropagation();
                      const id = menuFor;
                      setMenuFor(null);
                      if (label === '🎁 Mở quà') hit(id);
                      else miss();
                    }}
                  >
                    {label}
                  </button>
                </li>
              ))}
            </ul>
          )}
          {done && (
            <div className="absolute inset-0 grid place-items-center">
              <div className="flex items-center gap-3 rounded-3xl bg-white/90 p-4 font-bold">
                <Mochi mood="happy" size={64} animated={false} /> Tuyệt vời! 🎉
              </div>
            </div>
          )}
        </div>
      )}
      {tip && <p className="font-semibold text-rose-ink">{tip}</p>}
    </div>
  );
}
```

```tsx
import { useState, type MouseEvent as RME } from 'react';
import { vfs, type OS } from '@office-neko/shared';

export type MenuAction = 'open' | 'rename' | 'copy' | 'cut' | 'remove' | 'restore' | 'newFolder' | 'paste';

interface Props {
  os: OS;
  fs: vfs.Vfs;
  cwd: string;
  selected: string[];
  canPaste: boolean;
  renamingId: string | null;
  onSelect: (ids: string[]) => void;
  onOpen: (id: string) => void;
  onMenu: (action: MenuAction, id: string | null) => void;
  onRenameDone: (id: string, name: string | null) => void;
  onDragStartItem: (id: string) => void;
  onDropOn: (folderId: string) => void;
}

function iconFor(n: vfs.VNode): string {
  if (n.type === 'folder') return '📁';
  const ext = n.name.toLowerCase().split('.').pop();
  if (ext === 'jpg' || ext === 'png') return '🖼️';
  if (ext === 'pdf') return '📕';
  if (ext === 'docx') return '📘';
  if (ext === 'xlsx') return '📗';
  if (ext === 'pptx') return '📙';
  return '📄';
}

export function FileArea(p: Props) {
  const [menu, setMenu] = useState<{ id: string | null } | null>(null);
  const inTrash = p.cwd === p.fs.trashId;
  const items = vfs.childrenOf(p.fs, p.cwd);

  const openMenu = (e: RME, id: string | null) => {
    e.preventDefault();
    e.stopPropagation();
    if (id && !p.selected.includes(id)) p.onSelect([id]);
    setMenu({ id });
  };
  const choose = (action: MenuAction) => {
    const id = menu?.id ?? null;
    setMenu(null);
    p.onMenu(action, id);
  };

  const menuItems: [MenuAction, string][] = menu?.id
    ? inTrash
      ? [['restore', 'Khôi phục']]
      : [['open', 'Mở'], ['rename', 'Đổi tên'], ['copy', 'Sao chép'], ['cut', 'Cắt'], ['remove', 'Xoá']]
    : inTrash
      ? []
      : [['newFolder', 'Thư mục mới'], ['paste', 'Dán']];

  return (
    <div
      data-testid="file-area"
      className="relative min-h-56 flex-1 p-3"
      onClick={(e) => {
        if (e.target === e.currentTarget) {
          p.onSelect([]);
          setMenu(null);
        }
      }}
      onContextMenu={(e) => openMenu(e, null)}
    >
      {items.length === 0 && <p className="p-4 text-muted">{inTrash ? 'Thùng rác đang trống.' : 'Thư mục này đang trống.'}</p>}
      <ul className="grid grid-cols-[repeat(auto-fill,minmax(7rem,1fr))] gap-2">
        {items.map((n) => {
          const selected = p.selected.includes(n.id);
          return (
            <li key={n.id}>
              {p.renamingId === n.id ? (
                <div className="flex flex-col items-center gap-1 rounded-xl p-2">
                  <span className="text-4xl" aria-hidden>{iconFor(n)}</span>
                  <input
                    aria-label="Tên mới"
                    defaultValue={n.name}
                    autoFocus
                    onFocus={(e) => {
                      const dot = n.type === 'file' ? n.name.lastIndexOf('.') : -1;
                      e.currentTarget.setSelectionRange(0, dot > 0 ? dot : n.name.length);
                    }}
                    onKeyDown={(e) => {
                      e.stopPropagation();
                      if (e.key === 'Enter') p.onRenameDone(n.id, e.currentTarget.value);
                      if (e.key === 'Escape') p.onRenameDone(n.id, null);
                    }}
                    onBlur={(e) => p.onRenameDone(n.id, e.currentTarget.value)}
                    className="w-full rounded-md border-2 border-rose px-1 text-center text-sm"
                  />
                </div>
              ) : (
                <button
                  type="button"
                  data-path={vfs.pathOf(p.fs, n.id)}
                  aria-pressed={selected}
                  draggable
                  className={`flex w-full flex-col items-center gap-1 rounded-xl p-2 text-sm ${selected ? 'bg-sky/40 ring-2 ring-sky' : 'hover:bg-sky/15'}`}
                  onClick={(e) => {
                    e.stopPropagation();
                    setMenu(null);
                    const add = p.os === 'mac' ? e.metaKey : e.ctrlKey;
                    if (add) p.onSelect(selected ? p.selected.filter((x) => x !== n.id) : [...p.selected, n.id]);
                    else p.onSelect([n.id]);
                  }}
                  onDoubleClick={() => p.onOpen(n.id)}
                  onContextMenu={(e) => openMenu(e, n.id)}
                  onDragStart={(e) => {
                    p.onDragStartItem(n.id);
                    e.dataTransfer?.setData('text/plain', n.id);
                  }}
                  onDragOver={(e) => n.type === 'folder' && e.preventDefault()}
                  onDrop={(e) => {
                    e.preventDefault();
                    e.stopPropagation();
                    if (n.type === 'folder') p.onDropOn(n.id);
                  }}
                >
                  <span className="text-4xl" aria-hidden>{iconFor(n)}</span>
                  <span className="break-all">{n.name}</span>
                </button>
              )}
            </li>
          );
        })}
      </ul>
      {menu && menuItems.length > 0 && (
        <ul role="menu" className="card absolute left-4 top-4 z-20 w-48 p-2">
          {menuItems.map(([action, label]) => (
            <li key={action}>
              <button
                type="button"
                role="menuitem"
                disabled={action === 'paste' && !p.canPaste}
                onClick={() => choose(action)}
                className="w-full rounded-xl px-3 py-1.5 text-left hover:bg-peach/30 disabled:opacity-40"
              >
                {label}
              </button>
            </li>
          ))}
        </ul>
      )}
    </div>
  );
```

```tsx
import { useCallback, useEffect, useReducer, useRef, useState, type KeyboardEvent } from 'react';
import { desktopSim, isModPressed, vfs } from '@office-neko/shared';
import type { SimProps } from '../types';
import { Chrome } from './Chrome';
import { FileArea, type MenuAction } from './FileArea';

type S = desktopSim.DesktopState;

function crumbs(s: S): string {
  const root = s.os === 'win' ? 'Máy tính này' : 'Thư mục nhà';
  if (s.cwd === s.fs.trashId) return 'Thùng rác';
  const parts = vfs.pathOf(s.fs, s.cwd).split('/').filter(Boolean);
  return [root, ...parts].join(' › ');
}

export function DesktopSim({ initial, onState }: SimProps<S>) {
  const [state, dispatch] = useReducer(desktopSim.reduceDesktop, initial);
  const [renamingId, setRenamingId] = useState<string | null>(null);
  const [tip, setTip] = useState<string | null>(null);
  const pendingRename = useRef(false);
  const dragIds = useRef<string[]>([]);

  useEffect(() => {
    onState(state);
    if (pendingRename.current && state.selected.length === 1) {
      pendingRename.current = false;
      setRenamingId(state.selected[0]!);
    }
  }, [state, onState]);

  const os = state.os;
  const inTrash = state.cwd === state.fs.trashId;
  const sel = state.selected;
  const one = sel.length === 1 ? sel[0]! : null;

  const newFolder = useCallback(() => {
    pendingRename.current = true;
    dispatch({ type: 'mkdir', name: desktopSim.uniqueName(state.fs, state.cwd, os === 'win' ? 'Thư mục mới' : 'thư mục không tên') });
  }, [state.fs, state.cwd, os]);

  const open = (id: string) => {
    const n = state.fs.nodes[id];
    if (!n) return;
    if (n.type === 'file') setTip('Đây là file để luyện tập nên chưa mở được 😺');
    else {
      setTip(null);
      dispatch({ type: 'open', id });
    }
  };

  const onMenu = (action: MenuAction, id: string | null) => {
    const ids = id ? [id] : sel;
    switch (action) {
      case 'open': if (id) open(id); break;
      case 'rename': if (id) setRenamingId(id); break;
      case 'copy': dispatch({ type: 'copy', ids }); break;
      case 'cut': dispatch({ type: 'cut', ids }); break;
      case 'remove': dispatch({ type: 'remove', ids }); break;
      case 'restore': if (id) dispatch({ type: 'restore', id }); break;
      case 'newFolder': newFolder(); break;
      case 'paste': dispatch({ type: 'paste' }); break;
    }
  };

  const onKeyDown = (e: KeyboardEvent<HTMLDivElement>) => {
    if (renamingId) return;
    const mod = isModPressed(e, os);
    const k = e.key.toLowerCase();
    const handled = () => e.preventDefault();
    if (mod && e.shiftKey && k === 'n') return handled(), newFolder();
    if (mod && k === 'c' && sel.length) return handled(), dispatch({ type: 'copy', ids: sel });
    if (mod && k === 'x' && sel.length) return handled(), dispatch({ type: 'cut', ids: sel });
    if (mod && k === 'v') return handled(), dispatch({ type: 'paste' });
    if (mod && k === 'z') return handled(), dispatch({ type: 'undo' });
    if (os === 'win') {
      if (e.key === 'F2' && one) return handled(), setRenamingId(one);
      if (e.key === 'Delete' && sel.length) return handled(), dispatch({ type: 'remove', ids: sel });
      if (e.key === 'Backspace' || (e.altKey && e.key === 'ArrowUp')) return handled(), dispatch({ type: 'up' });
      if (e.key === 'Enter' && one) return handled(), open(one);
    } else {
      if (e.key === 'Enter' && one) return handled(), setRenamingId(one);
      if (mod && e.key === 'Backspace' && sel.length) return handled(), dispatch({ type: 'remove', ids: sel });
      if (mod && e.key === 'ArrowUp') return handled(), dispatch({ type: 'up' });
      if (mod && (e.key === 'ArrowDown' || k === 'o') && one) return handled(), open(one);
    }
  };

  const btn = 'rounded-lg px-2 py-1 text-sm hover:bg-sky/30 disabled:opacity-40';
  const topFolders = vfs.childrenOf(state.fs, state.fs.rootId).filter((n) => n.type === 'folder');

  return (
    <div data-testid="desktop-sim" tabIndex={0} onKeyDown={onKeyDown} className="outline-none focus-visible:ring-4 focus-visible:ring-sky rounded-3xl">
      <Chrome
        os={os}
        onTrash={() => dispatch({ type: 'open', id: state.fs.trashId })}
        onDropToTrash={() => {
          if (dragIds.current.length) dispatch({ type: 'remove', ids: dragIds.current });
          dragIds.current = [];
        }}
      >
        <div className="flex flex-wrap items-center gap-1 border-b border-muted/20 px-2 py-1">
          {inTrash ? (
            <>
              <button type="button" className={btn} data-help-id="desktop.restore" disabled={!one} onClick={() => one && dispatch({ type: 'restore', id: one })}>♻️ Khôi phục</button>
              <button type="button" className={btn} data-help-id="desktop.emptyTrash" onClick={() => dispatch({ type: 'emptyTrash' })}>🧹 Dọn sạch thùng rác</button>
            </>
          ) : (
            <>
              <button type="button" className={btn} data-help-id="desktop.newFolder" onClick={newFolder}>📁 Thư mục mới</button>
              <button type="button" className={btn} data-help-id="desktop.rename" disabled={!one} onClick={() => one && setRenamingId(one)}>✏️ Đổi tên</button>
              <button type="button" className={btn} data-help-id="desktop.delete" disabled={!sel.length} onClick={() => dispatch({ type: 'remove', ids: sel })}>🗑️ Xoá</button>
              <button type="button" className={btn} data-help-id="desktop.copy" disabled={!sel.length} onClick={() => dispatch({ type: 'copy', ids: sel })}>📋 Sao chép</button>
              <button type="button" className={btn} data-help-id="desktop.cut" disabled={!sel.length} onClick={() => dispatch({ type: 'cut', ids: sel })}>✂️ Cắt</button>
              <button type="button" className={btn} data-help-id="desktop.paste" disabled={!state.clipboard} onClick={() => dispatch({ type: 'paste' })}>📥 Dán</button>
              <button type="button" className={btn} data-help-id="desktop.undo" disabled={!state.past.length} onClick={() => dispatch({ type: 'undo' })}>↩️ Hoàn tác</button>
            </>
          )}
          <button type="button" className={btn} data-help-id="desktop.up" onClick={() => dispatch({ type: 'up' })}>⬆️ Lên</button>
        </div>
        <div aria-label="Thanh địa chỉ" data-help-id="desktop.address" className="border-b border-muted/20 bg-cream px-3 py-1 text-sm">
          {crumbs(state)}
        </div>
        <div className="flex min-h-64">
          <nav aria-label="Thanh bên" className="w-40 shrink-0 space-y-1 border-r border-muted/20 p-2 text-sm">
            <button type="button" className={`${btn} w-full text-left`} onClick={() => dispatch({ type: 'open', id: state.fs.rootId })}>
              🏠 {os === 'win' ? 'Máy tính này' : 'Thư mục nhà'}
            </button>
            {topFolders.map((f) => (
              <button key={f.id} type="button" className={`${btn} w-full text-left`} onClick={() => dispatch({ type: 'open', id: f.id })}>
                📁 {f.name}
              </button>
            ))}
            <button
              type="button"
              className={`${btn} w-full text-left`}
              onClick={() => dispatch({ type: 'open', id: state.fs.trashId })}
              onDragOver={(e) => e.preventDefault()}
              onDrop={() => {
                if (dragIds.current.length) dispatch({ type: 'remove', ids: dragIds.current });
                dragIds.current = [];
              }}
            >
              🗑️ Thùng rác
            </button>
          </nav>
          <FileArea
            os={os}
            fs={state.fs}
            cwd={state.cwd}
            selected={sel}
            canPaste={!!state.clipboard}
            renamingId={renamingId}
            onSelect={(ids) => dispatch({ type: 'select', ids })}
            onOpen={open}
            onMenu={onMenu}
            onRenameDone={(id, name) => {
              setRenamingId(null);
              const cur = state.fs.nodes[id]?.name;
              if (name !== null && name.trim() && name.trim() !== cur) dispatch({ type: 'rename', id, name });
            }}
            onDragStartItem={(id) => {
              dragIds.current = sel.includes(id) ? sel : [id];
            }}
            onDropOn={(folderId) => {
              const ids = dragIds.current.filter((x) => x !== folderId);
              if (ids.length) dispatch({ type: 'move', ids, destId: folderId });
              dragIds.current = [];
            }}
          />
        </div>
        {state.error && (
          <div role="alert" className="flex items-center gap-3 border-t border-rose/40 bg-rose/10 px-3 py-2 text-sm text-rose-ink">
            <span className="flex-1">{vfs.VFS_ERROR_VI[state.error]}</span>
            <button type="button" className="btn !px-3 !py-1" onClick={() => dispatch({ type: 'clearError' })}>Đã hiểu</button>
          </div>
        )}
        {tip && <p className="border-t border-muted/20 px-3 py-2 text-sm text-muted">{tip}</p>}
      </Chrome>
    </div>
  );
}
```

### Task 9: Giao diện gõ phím (đoạn văn tô màu, bàn phím ảo, cảnh báo bộ gõ)

**Files:**
- Create: `client/src/sims/keyboard/OnScreenKeyboard.tsx`, `client/src/sims/keyboard/KeyboardSim.tsx`, `client/src/sims/keyboard/KeyboardSim.test.tsx`
- Modify: `client/src/sims/registry.ts`, `client/src/content/help-ids.ts`

**Interfaces:**
- Consumes: `typingSim` namespace (Task 5), `KEY_ROWS`, `SPACE_KEY`, `FINGER_VI`, `FINGER_COLOR`, `keyForChar`, `keyDef`, `compareText` (shared).
- Produces:
  - `OnScreenKeyboard({ os, next }: { os: OS; next: string | null })`. `next` là ký tự cần gõ kế tiếp. Phím cần bấm có `data-next="true"`; nếu cần Shift thì phím Shift ở tay bên kia cũng có `data-next="true"`. Mỗi phím có `data-key={id}`. Bên dưới có dòng "Dùng: <ngón>".
  - `KeyboardSim(props: SimProps<typingSim.TypingState>)`
  - `SIMS.keyboard = { kind: 'keyboard', Component: KeyboardSim, reduce: typingSim.reduceTyping }`
  - HELP_IDS: `keyboard.area` ("Vùng gõ phím"), `keyboard.board` ("Bàn phím ảo").
- **Hành vi:**
  - **Vùng gõ:** `role="textbox"`, `aria-label="Vùng gõ phím"`, `tabIndex=0`, tự focus.
  - **Xử lý `keyDown`:**
    - Bỏ qua khi có Ctrl/⌘/Alt.
    - Nếu nhận `key === 'Process'`, `nativeEvent.isComposing`, `keyCode === 229`, hoặc 1 ký tự không phải ASCII → hiện cảnh báo bộ gõ (`role="alert"`) và không gửi phím vào sim.
    - `Backspace` hoặc phím 1 ký tự → `preventDefault` rồi dispatch `{ type: 'key', key, at: Date.now() }`.
  - **Tô màu từng ký tự:** mỗi ký tự là `<span data-state=…>` với các trạng thái:
    - `correct`: đúng, chữ xanh.
    - `wrong`: sai, nền đỏ nhạt, chữ đỏ đậm.
    - `composing`: đang gõ dở trong từ (chỉ ở chế độ Telex/VNI), gạch chân chấm.
    - `pending`: chưa gõ, chữ xám.
    - Ký tự ngay tại con trỏ có thêm `data-cursor="true"` và viền dưới.
  - **Thanh số liệu:** "WPM", "Chính xác", và "Còn lại Xs" khi có `durationSec`. Khi có `durationSec` và đã bắt đầu gõ, cứ 250 ms dispatch `tick` một lần.

```tsx
import { useEffect, useReducer, useRef, useState, type KeyboardEvent } from 'react';
import { compareText, typingSim } from '@office-neko/shared';
import type { SimProps } from '../types';
import { Mochi } from '../../components/Mochi';
import { OnScreenKeyboard } from './OnScreenKeyboard';

type S = typingSim.TypingState;

export function KeyboardSim({ initial, onState, os }: SimProps<S>) {
  const [state, dispatch] = useReducer(typingSim.reduceTyping, initial);
  const [imeWarning, setImeWarning] = useState(false);
  const boxRef = useRef<HTMLDivElement>(null);

  useEffect(() => {
    boxRef.current?.focus();
  }, []);

  useEffect(() => {
    onState(state);
  }, [state, onState]);

  useEffect(() => {
    if (state.durationSec === null || state.startedAt === null || state.finished) return;
    const id = window.setInterval(() => dispatch({ type: 'tick', at: Date.now() }), 250);
    return () => window.clearInterval(id);
  }, [state.durationSec, state.startedAt, state.finished]);

  const onKeyDown = (e: KeyboardEvent<HTMLDivElement>) => {
    if (e.ctrlKey || e.metaKey || e.altKey) return;
    const composing = e.key === 'Process' || e.nativeEvent.isComposing || e.keyCode === 229;
    if (composing || (e.key.length === 1 && /[^\x20-\x7e]/.test(e.key))) {
      e.preventDefault();
      setImeWarning(true);
      return;
    }
    if (e.key === 'Backspace' || e.key.length === 1) {
      e.preventDefault();
      dispatch({ type: 'key', key: e.key, at: Date.now() });
    }
  };

  const text = typingSim.typedText(state);
  const states = compareText(state.target, text);
  const committed = state.ime ? [...state.imeState.done].length : Infinity;
  const textLen = [...text].length;
  const targetChars = [...state.target];
  const m = typingSim.typingMetrics(state);
  const next = state.ime || state.finished ? null : (targetChars[textLen] ?? null);
  const left =
    state.durationSec !== null
      ? Math.max(0, Math.ceil(state.durationSec - (state.startedAt !== null && state.lastAt !== null ? (Date.now() - state.startedAt) / 1000 : 0)))
      : null;

  return (
    <div className="card space-y-3 p-4">
      <div className="flex flex-wrap gap-4 text-sm font-bold">
        <span>⚡ WPM: {m.wpm}</span>
        <span>🎯 Chính xác: {m.accuracy}%</span>
        {left !== null && <span>⏱️ Còn lại {left}s</span>}
      </div>
      {imeWarning && (
        <p role="alert" className="rounded-2xl bg-butter/50 p-3 text-sm">
          Có vẻ bộ gõ tiếng Việt của máy {os === 'mac' ? '(Vietnamese Telex trên thanh menu)' : '(Unikey/EVKey hoặc bàn phím Vietnamese)'} đang bật. Hãy tắt nó đi — Mochi sẽ tự thêm dấu cho bạn.
        </p>
      )}
      <div
        role="textbox"
        aria-label="Vùng gõ phím"
        data-help-id="keyboard.area"
        ref={boxRef}
        tabIndex={0}
        onKeyDown={onKeyDown}
        className="rounded-2xl bg-white p-4 font-mono text-2xl leading-relaxed tracking-wide outline-none ring-2 ring-peach/50 focus:ring-4 focus:ring-sky"
      >
        {targetChars.map((ch, i) => {
          let st: string = states[i] ?? 'pending';
          if (state.ime && i >= committed && i < textLen) st = 'composing';
          const cls =
            st === 'correct' ? 'text-excel-ink' : st === 'wrong' ? 'bg-rose/30 text-rose-ink' : st === 'composing' ? 'underline decoration-dotted' : 'text-muted';
          const cursor = i === textLen && !state.finished;
          return (
            <span key={i} data-state={st} data-cursor={cursor || undefined} className={`${cls} ${cursor ? 'border-b-4 border-rose' : ''}`}>
              {ch}
            </span>
          );
        })}
      </div>
      {state.finished ? (
        <div className="flex items-center gap-3 rounded-2xl bg-mint/30 p-3 font-bold">
          <Mochi mood="happy" size={56} animated={false} />
          {state.durationSec !== null && !m.complete ? 'Hết giờ!' : 'Xong rồi! 🎉'} {m.wpm} WPM · {m.accuracy}% chính xác
        </div>
      ) : state.ime ? (
        <p className="rounded-2xl bg-lavender/30 p-3 text-sm">
          ✍️ Kiểu gõ {state.ime === 'telex' ? 'Telex: aa → â, dd → đ, w → ư/ơ/ă, s f r x j → sắc huyền hỏi ngã nặng' : 'VNI: 1–5 → dấu, 6 → mũ, 7 → móc, 8 → trăng, 9 → đ'}
        </p>
      ) : null}
      {!state.ime && <OnScreenKeyboard os={os} next={next} />}
    </div>
  );
}
```

```ts
import { desktopSim, mouseSim, typingSim, type ImeMethod, type OS, type PracticeStep, type Spotlight, type TreeSpec } from '@office-neko/shared';

export function keyActions(raw: string, msPerKey = 150, start = 1000): typingSim.TypingAction[] {
  return [...raw].map((key, i) => ({ type: 'key', key, at: start + i * msPerKey }));
}

export function mousePractice(o: {
  game: mouseSim.MouseGame;
  total: number;
  task: string;
  hint: string;
  steps: string[];
}): PracticeStep<mouseSim.MouseState, mouseSim.MouseCheck, mouseSim.MouseAction> {
  const initial = mouseSim.mouseInitial(o.game, o.total);
  return {
    kind: 'practice',
    sim: 'mouse',
    task: o.task,
    initial,
    checks: [{ kind: 'mouseGame', minHits: o.total }],
    hints: [o.hint, { helpId: o.game === 'drag' ? 'mouse.basket' : 'mouse.target' }, o.steps],
    solution: mouseSim.mouseTargets(initial).map((id) => ({ type: 'hit', id })),
  };
}

export function typingPractice(o: {
  task: string;
  target: string;
  raw?: string;
  ime?: ImeMethod | null;
  mode?: typingSim.TypingMode;
  allowBackspace?: boolean;
  checks?: typingSim.TypingCheck[];
  hint: string;
  steps: string[];
}): PracticeStep<typingSim.TypingState, typingSim.TypingCheck, typingSim.TypingAction> {
  return {
    kind: 'practice',
    sim: 'keyboard',
    task: o.task,
    initial: typingSim.typingInitial({ target: o.target, ime: o.ime ?? null, mode: o.mode ?? 'drill', allowBackspace: o.allowBackspace ?? true }),
    checks: o.checks ?? [{ kind: 'typing', completeText: true }],
    hints: [o.hint, { helpId: 'keyboard.area' }, o.steps],
    solution: keyActions(o.raw ?? o.target),
  };
}

export function desktopPractice(o: {
  os: OS;
  tree: TreeSpec[];
  cwd?: string;
  task: string;
  checks: desktopSim.DesktopCheck[];
  spotlight: Spotlight;
  hint: string;
  steps: string[];
  /** Thao tác chuẩn bị trước (vd. bỏ sẵn một file vào thùng rác); không tính vào lịch sử hoàn tác. */
  setup?: (s0: desktopSim.DesktopState) => desktopSim.DesktopAction[];
  solve: (s0: desktopSim.DesktopState) => desktopSim.DesktopAction[];
}): PracticeStep<desktopSim.DesktopState, desktopSim.DesktopCheck, desktopSim.DesktopAction> {
  const fresh = desktopSim.desktopInitial(o.tree, o.os, o.cwd ?? '/');
  const prepared = (o.setup ? o.setup(fresh) : []).reduce(desktopSim.reduceDesktop, fresh);
  const initial = { ...prepared, past: [], selected: [], clipboard: null };
  return {
    kind: 'practice',
    sim: 'desktop',
    task: o.task,
    initial,
    checks: o.checks,
    hints: [o.hint, o.spotlight, o.steps],
    solution: o.solve(initial),
  };
}
```

```ts
import { useLiveQuery } from 'dexie-react-hooks';
import { dayKey, type KeyStats } from '@office-neko/shared';
import { db, type KeyStatRow, type TypingRun } from '../db/db';

export async function recordTypingRun(r: {
  mode: string;
  wpm: number;
  accuracy: number;
  elapsedMs: number;
  keyStats: KeyStats;
  now?: Date;
}): Promise<void> {
  const now = r.now ?? new Date();
  await db.transaction('rw', db.typingRuns, db.keyStats, db.activity, async () => {
    await db.typingRuns.add({ at: now.toISOString(), mode: r.mode, wpm: r.wpm, accuracy: r.accuracy });
    for (const [key, s] of Object.entries(r.keyStats)) {
      const prev = await db.keyStats.get(key);
      await db.keyStats.put({ key, hit: (prev?.hit ?? 0) + s.hit, miss: (prev?.miss ?? 0) + s.miss });
    }
    const day = dayKey(now);
    const act = (await db.activity.get(day)) ?? { day, minutes: 0, xp: 0 };
    await db.activity.put({ ...act, minutes: act.minutes + Math.max(1, Math.round(r.elapsedMs / 60_000)) });
  });
}

const NO_RUNS: TypingRun[] = [];
const NO_STATS: KeyStatRow[] = [];

export function useTypingRuns(limit = 20): TypingRun[] {
  const rows = useLiveQuery(() => db.typingRuns.orderBy('at').reverse().limit(limit).toArray(), [limit], NO_RUNS);
  return [...rows].reverse();
}

export function useKeyStats(): KeyStats {
  const rows = useLiveQuery(() => db.keyStats.toArray(), [], NO_STATS);
  return Object.fromEntries(rows.map((r) => [r.key, { hit: r.hit, miss: r.miss }]));
}
```

```tsx
import type { TypingRun } from '../db/db';

export function WpmChart({ runs }: { runs: TypingRun[] }) {
  if (runs.length === 0) return <p className="text-muted">Chưa có lần kiểm tra nào. Thử một bài 1 phút nhé!</p>;
  const W = 320;
  const H = 120;
  const max = Math.max(40, ...runs.map((r) => r.wpm));
  const x = (i: number) => (runs.length === 1 ? W / 2 : 16 + (i * (W - 32)) / (runs.length - 1));
  const y = (wpm: number) => H - 16 - (wpm / max) * (H - 32);
  const points = runs.map((r, i) => `${x(i)},${y(r.wpm)}`).join(' ');
  return (
    <svg role="img" aria-label="Biểu đồ WPM" viewBox={`0 0 ${W} ${H}`} className="w-full max-w-md">
      <line x1="16" y1={H - 16} x2={W - 16} y2={H - 16} stroke="#E6DDD3" />
      <polyline points={points} fill="none" stroke="#D6336C" strokeWidth="3" strokeLinejoin="round" />
      {runs.map((r, i) => (
        <g key={i}>
          <circle cx={x(i)} cy={y(r.wpm)} r="4" fill="#D6336C" />
          <text x={x(i)} y={y(r.wpm) - 8} textAnchor="middle" fontSize="10" fill="#3B2E4A">{r.wpm}</text>
        </g>
      ))}
    </svg>
  );
```

```tsx
import { useCallback, useEffect, useRef, useState } from 'react';
import { Link, Navigate, useParams } from 'react-router-dom';
import { typingSim } from '@office-neko/shared';
import { useOs } from '../db/profile';
import { speedTestText, TYPING_TEXTS } from '../content/typing-texts';
import { KeyboardSim } from '../sims/keyboard/KeyboardSim';
import { recordTypingRun } from '../progress/typing';
import { Mochi } from '../components/Mochi';

function makeInitial(minutes: number, attempt: number) {
  return typingSim.typingInitial({
    target: speedTestText(minutes, attempt % TYPING_TEXTS.length),
    mode: 'test',
    durationSec: minutes * 60,
  });
}

export function SpeedTestPage() {
  const { min = '' } = useParams();
  const minutes = min === '1' ? 1 : min === '3' ? 3 : null;
  const os = useOs();
  const [attempt, setAttempt] = useState(0);
  const [initial, setInitial] = useState(() => makeInitial(minutes ?? 1, 0));
  const [result, setResult] = useState<ReturnType<typeof typingSim.typingMetrics> | null>(null);
  const saved = useRef(false);

  const onState = useCallback(
    (s: typingSim.TypingState) => {
      if (!s.finished || saved.current || minutes === null) return;
      saved.current = true;
      const m = typingSim.typingMetrics(s);
      setResult(m);
      void recordTypingRun({ mode: `test-${minutes}`, wpm: m.wpm, accuracy: m.accuracy, elapsedMs: m.elapsedMs, keyStats: s.keyStats }).catch(() => undefined);
    },
    [minutes],
  );

  useEffect(() => {
    saved.current = false;
    setResult(null);
    if (minutes !== null) setInitial(makeInitial(minutes, attempt));
  }, [attempt, minutes]);

  if (minutes === null) return <Navigate to="/typing" replace />;

  return (
    <div className="mx-auto max-w-4xl space-y-4 p-4 pb-24">
      <div className="flex items-center gap-3">
        <Link to="/typing" className="btn">← Luyện gõ</Link>
        <h1 className="text-2xl font-extrabold">Kiểm tra tốc độ {minutes} phút</h1>
      </div>
      <p className="text-muted">Đồng hồ bắt đầu chạy khi bạn gõ phím đầu tiên. Gõ sai có thể dùng Backspace để sửa.</p>
      <KeyboardSim key={attempt} initial={initial} onState={onState} os={os} spotlight={null} />
      {result && (
        <section className="card flex flex-wrap items-center gap-4 p-5">
          <Mochi mood="happy" size={80} />
          <div className="flex-1 space-y-1">
            <h2 className="text-xl font-extrabold">Kết quả</h2>
            <p className="text-lg font-bold">{result.wpm} WPM · {result.accuracy}% chính xác</p>
          </div>
          <button type="button" className="btn-primary" onClick={() => setAttempt((a) => a + 1)}>Làm lại</button>
        </section>
      )}
    </div>
  );
}
```

`client/src/pages/FishGamePage.tsx`:
```tsx
import { useEffect, useReducer, useRef, useState } from 'react';
import { Link } from 'react-router-dom';
import { fishGame } from '@office-neko/shared';
import { FISH_WORDS } from '../content/fish-words';
import { Mochi } from '../components/Mochi';

function Game({ onRestart }: { onRestart: () => void }) {
  const [state, dispatch] = useReducer(fishGame.reduceFish, undefined, () => fishGame.fishInitial(FISH_WORDS, Date.now() >>> 0));
  const [paused, setPaused] = useState(true);
  const [started, setStarted] = useState(false);
  const last = useRef<number | null>(null);

  useEffect(() => {
    if (paused || state.over) return;
    let alive = true;
    let handle = 0;
    const raf = window.requestAnimationFrame?.bind(window) ?? ((cb: FrameRequestCallback) => window.setTimeout(() => cb(performance.now()), 16));
    const cancel = window.cancelAnimationFrame?.bind(window) ?? window.clearTimeout;
    const frame = (now: number) => {
      if (!alive) return;
      if (!document.hidden) {
        const dt = last.current === null ? 16 : Math.min(100, now - last.current);
        dispatch({ type: 'tick', dt });
      }
      last.current = now;
      handle = raf(frame);
    };
    handle = raf(frame);
    return () => {
      alive = false;
      last.current = null;
      cancel(handle);
    };
  }, [paused, state.over]);

  useEffect(() => {
    const onKey = (e: KeyboardEvent) => {
      if (!started || state.over) return;
      if (e.key === ' ') {
        e.preventDefault();
        setPaused((p) => !p);
        return;
      }
      if (paused || e.ctrlKey || e.metaKey || e.altKey) return;
      if (e.key.length === 1 || e.key === 'Backspace' || e.key === 'Escape' || e.key === 'Enter') {
        e.preventDefault();
        dispatch({ type: 'key', key: e.key });
      }
    };
    window.addEventListener('keydown', onKey);
    return () => window.removeEventListener('keydown', onKey);
  }, [started, paused, state.over]);

  const target = fishGame.targetFish(state);
  const catX = target ? target.x : 0.5;

  return (
    <div className="space-y-3">
      <div className="card flex flex-wrap items-center gap-4 p-3 font-bold">
        <span>Điểm: {state.score}</span>
        <span>Cấp: {state.level}</span>
        <span aria-label={`${state.lives} mạng`}>{'❤️'.repeat(Math.max(0, state.lives))}</span>
        <span className="ml-auto rounded-full bg-cream px-3 py-1 font-mono" aria-label="Chữ đang gõ">{state.input}</span>
        {started && !state.over && (
          <button type="button" className="btn-soft !py-1" onClick={() => setPaused((p) => !p)}>
            {paused ? '▶️ Chơi tiếp' : '⏸️ Tạm dừng'}
          </button>
        )}
      </div>
      <div className="relative h-[28rem] overflow-hidden rounded-3xl bg-gradient-to-b from-sky/60 to-mint/50">
        {state.fish.map((f) => {
          const isTarget = target?.id === f.id;
          const typed = isTarget ? state.input.length : 0;
          return (
            <div key={f.id} className="absolute -translate-x-1/2 text-center" style={{ left: `${f.x * 100}%`, top: `${f.y * 88}%` }}>
              <div className="text-3xl">🐟</div>
              <div className={`rounded-full bg-white px-2 font-mono font-bold ${isTarget ? 'ring-2 ring-rose' : ''}`}>
                <span className="text-excel-ink">{f.word.slice(0, typed)}</span>
                {f.word.slice(typed)}
              </div>
            </div>
          );
        })}
        <div className="absolute bottom-1 -translate-x-1/2 transition-all" style={{ left: `${catX * 100}%` }}>
          <Mochi mood={state.over ? 'sad' : 'happy'} size={56} animated={false} />
        </div>
        {!started && (
          <div className="absolute inset-0 grid place-items-center bg-white/70 p-6 text-center">
            <div className="space-y-3">
              <p className="text-lg font-bold">Gõ đúng chữ trên con cá để Mochi bắt nó trước khi cá rơi xuống đáy!</p>
              <p className="text-muted">Phím Space: tạm dừng. Esc: xoá chữ đang gõ.</p>
              <button
                type="button"
                className="btn-primary"
                onClick={() => {
                  setStarted(true);
                  setPaused(false);
                }}
              >
                Bắt đầu
              </button>
            </div>
          </div>
        )}
        {state.over && (
          <div className="absolute inset-0 grid place-items-center bg-white/80 p-6 text-center">
            <div className="card space-y-3 p-6">
              <h2 className="text-2xl font-extrabold">Hết lượt!</h2>
              <p>Điểm: {state.score} · Bắt được {state.caught} con cá</p>
              <div className="flex justify-center gap-3">
                <button type="button" className="btn-primary" onClick={onRestart}>Chơi lại</button>
                <Link to="/typing" className="btn-soft">← Luyện gõ</Link>
              </div>
            </div>
          </div>
        )}
      </div>
    </div>
  );
}

export function FishGamePage() {
  const [round, setRound] = useState(0);
  return (
    <div className="mx-auto max-w-4xl space-y-4 p-4 pb-24">
      <div className="flex items-center gap-3">
        <Link to="/typing" className="btn">← Luyện gõ</Link>
        <h1 className="text-2xl font-extrabold">🐟 Mèo bắt cá</h1>
      </div>
      <Game key={round} onRestart={() => setRound((r) => r + 1)} />
    </div>
  );
}
```
